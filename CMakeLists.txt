cmake_minimum_required(VERSION 3.25)
project(Murasaki
  VERSION 0.1.1
  DESCRIPTION "Animated wallpapers and desktop overlays for Linux"
  LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)

option(MURASAKI_WERROR "Treat compiler warnings as errors" OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
if(MURASAKI_WERROR)
  add_compile_options(-Werror)
endif()

find_package(Qt6 6.8 REQUIRED COMPONENTS Core DBus Network)
include(CTest)
include(GNUInstallDirs)

# murasaki-surface and the Murasaki.Render QML module need the GUI stack; core, daemon and ctl never do.
set(MURASAKI_BUILD_SURFACE AUTO CACHE STRING "Build murasaki-surface and Murasaki.Render: AUTO, ON or OFF")
set_property(CACHE MURASAKI_BUILD_SURFACE PROPERTY STRINGS AUTO ON OFF)
if(NOT MURASAKI_BUILD_SURFACE MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "MURASAKI_BUILD_SURFACE must be AUTO, ON or OFF, not '${MURASAKI_BUILD_SURFACE}'")
endif()
set(MURASAKI_SURFACE OFF)
if(NOT MURASAKI_BUILD_SURFACE STREQUAL "OFF")
  find_package(Qt6 6.8 QUIET COMPONENTS Gui OpenGL Qml Quick Network WebEngineCore WebEngineQuick)
  find_package(PkgConfig)
  if(PkgConfig_FOUND)
    pkg_check_modules(MPV IMPORTED_TARGET mpv)
    pkg_check_modules(PIPEWIRE IMPORTED_TARGET libpipewire-0.3)
    pkg_check_modules(EGL IMPORTED_TARGET egl)
    # The overlay decodes pet animations into GPU textures (src/overlay/spritedecoder.cpp).
    pkg_check_modules(FFMPEG IMPORTED_TARGET libavformat libavcodec libswscale libavutil)
    pkg_check_modules(WEBPDEMUX IMPORTED_TARGET libwebpdemux libwebp) # animated WebP pets (the optional compositing decoder)
    pkg_check_modules(GBM IMPORTED_TARGET gbm)
    pkg_check_modules(XCB IMPORTED_TARGET xcb) # the X11 presenter
  endif()
  if(Qt6Gui_FOUND AND Qt6OpenGL_FOUND AND Qt6Qml_FOUND AND Qt6Quick_FOUND AND Qt6Network_FOUND
     AND Qt6WebEngineCore_FOUND AND Qt6WebEngineQuick_FOUND AND MPV_FOUND AND EGL_FOUND)
    set(MURASAKI_SURFACE ON)
    find_package(LayerShellQt 6 QUIET)
  elseif(MURASAKI_BUILD_SURFACE STREQUAL "ON")
    message(FATAL_ERROR "MURASAKI_BUILD_SURFACE=ON needs Qt 6 Gui, OpenGL, Qml, Quick, Network, WebEngineQuick, libmpv and EGL")
  else()
    message(STATUS "murasaki-surface disabled: Qt Quick, Qt WebEngine, libmpv or EGL not found")
  endif()
endif()

# The murasaki GUI needs Qt >= 6.8 (QML loadFromModule, WebEngine permission requests and persistent
# permission policy) with Qt Quick, Quick Controls 2, Widgets and WebEngine; Kirigami is a run-time import.
set(MURASAKI_APP_MIN_QT 6.8)
set(MURASAKI_BUILD_APP AUTO CACHE STRING "Build the murasaki GUI: AUTO, ON or OFF")
set_property(CACHE MURASAKI_BUILD_APP PROPERTY STRINGS AUTO ON OFF)
if(NOT MURASAKI_BUILD_APP MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "MURASAKI_BUILD_APP must be AUTO, ON or OFF, not '${MURASAKI_BUILD_APP}'")
endif()
set(MURASAKI_APP OFF)
if(NOT MURASAKI_BUILD_APP STREQUAL "OFF")
  if(Qt6Core_VERSION VERSION_LESS MURASAKI_APP_MIN_QT)
    if(MURASAKI_BUILD_APP STREQUAL "ON")
      message(FATAL_ERROR "MURASAKI_BUILD_APP=ON needs Qt ${MURASAKI_APP_MIN_QT} or newer, found ${Qt6Core_VERSION}")
    endif()
    message(STATUS "murasaki GUI disabled: needs Qt ${MURASAKI_APP_MIN_QT} or newer, found ${Qt6Core_VERSION}")
  else()
  find_package(Qt6 ${MURASAKI_APP_MIN_QT} QUIET COMPONENTS Widgets Qml Quick QuickControls2 WebEngineQuick)
  if(Qt6Widgets_FOUND AND Qt6Qml_FOUND AND Qt6Quick_FOUND AND Qt6QuickControls2_FOUND AND Qt6WebEngineQuick_FOUND)
    set(MURASAKI_APP ON)
  elseif(MURASAKI_BUILD_APP STREQUAL "ON")
    message(FATAL_ERROR "MURASAKI_BUILD_APP=ON needs Qt 6 Widgets, Qml, Quick, QuickControls2 and WebEngineQuick")
  else()
    message(STATUS "murasaki GUI disabled: Qt Widgets, Quick Controls 2 or WebEngineQuick not found")
  endif()
  endif()
endif()

# The own Vulkan scene engine (murasaki-scene-vk): plain C++20, no Qt. It needs the Vulkan headers and lz4; volk,
# VMA and stb_image are vendored and glslang is fetched (or taken from the system), see src/scene-vk/CMakeLists.txt.
set(MURASAKI_SCENE_VK AUTO CACHE STRING "Build murasaki-scene-vk, the Vulkan scene engine: AUTO, ON or OFF")
set_property(CACHE MURASAKI_SCENE_VK PROPERTY STRINGS AUTO ON OFF)
if(NOT MURASAKI_SCENE_VK MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "MURASAKI_SCENE_VK must be AUTO, ON or OFF, not '${MURASAKI_SCENE_VK}'")
endif()

add_subdirectory(src/bridge)
add_subdirectory(src/core)
add_subdirectory(src/import)
add_subdirectory(src/media)
add_subdirectory(src/audio)
add_subdirectory(src/sandbox)
add_subdirectory(src/daemon)
add_subdirectory(src/ctl)
add_subdirectory(src/render)
add_subdirectory(src/x11)
add_subdirectory(src/surface)
add_subdirectory(src/shortcut)
add_subdirectory(src/app)
add_subdirectory(src/overlay)
if(NOT MURASAKI_SCENE_VK STREQUAL "OFF")
  add_subdirectory(src/scene-vk)
endif()
add_subdirectory(src/puppet)
add_subdirectory(src/screenmatch)
add_subdirectory(data)
include(cmake/Packaging.cmake)
if(TARGET murasaki_dmabuf AND Qt6Gui_FOUND)
  add_subdirectory(tools/bridge-view)
endif()
if(TARGET murasaki-overlay AND BUILD_TESTING)
  add_subdirectory(tools/overlay-backdrop)
  add_subdirectory(tools/overlay-pointer-probe)
  add_subdirectory(tools/stage-probe)
endif()
option(MURASAKI_BUILD_FUZZ "Build the fuzz drivers of the import parsers (use a sanitizer build directory)" OFF)
if(MURASAKI_BUILD_FUZZ)
  add_subdirectory(tools/fuzz)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_subdirectory(tests)
endif()
